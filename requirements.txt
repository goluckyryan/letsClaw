# letsClaw — Python dependencies
#
# Install:  python3 -m venv .venv && .venv/bin/pip install -r requirements.txt
#
# The versions below are what the project runs on (Python 3.14). They are the
# floor, not a hard cap — anything newer that keeps the same API works, but
# these are the ones that are known to.

# --- required for the core, the WebUI, and the terminal client ---------------
aiohttp==3.14.3        # the WS/HTTP server (server.py) and every client
openai==3.24.0         # llm_engine.py — the OpenAI-compatible chat API
PyYAML==6.0.3          # config.yaml (imported as `yaml`)
# Token accounting. token_counter.py counts with cl100k_base, and the
# corrections that decide when a window rolls over are calibrated against it.
# There is a chars/4 fallback so a missing tiktoken never stops the core, but
# it is a much cruder count and the calibration does not apply to it — so
# install this. First use fetches a 1.6 MB vocab file into state/tiktoken/,
# once, and never needs the network again.
tiktoken==0.14.0

# --- optional ----------------------------------------------------------------
# The Discord bot (source/discord_client.py) only. The core never imports
# discord.py and does not know the bot exists, so skip this if you do not run
# a bot.
discord.py==2.7.1
